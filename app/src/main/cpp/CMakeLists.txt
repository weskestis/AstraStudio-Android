cmake_minimum_required(VERSION 3.22.1)
project(astra_audio LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(astra_core STATIC core/RenderCore.cpp dsp/SynthPatch.cpp dsp/WavetableBank.cpp dsp/Filter.cpp dsp/SynthVoice.cpp
    fx/FxConfig.cpp fx/FxPrimitives.cpp fx/FxModules.cpp fx/FxRack.cpp fx/Spectrum.cpp)
set_target_properties(astra_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(astra_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(astra_core PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
target_compile_options(astra_core PRIVATE $<$<CONFIG:Debug>:-O2>)

if(ANDROID)
    find_package(oboe REQUIRED CONFIG)
    add_library(astra_audio SHARED AudioEngine.cpp native-lib.cpp)
    target_link_libraries(astra_audio PRIVATE astra_core oboe::oboe android log)
    target_compile_options(astra_audio PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(astra_audio PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
else()
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(astra_core_tests tests/CoreTests.cpp tests/AllocationTracker.cpp)
    target_link_libraries(astra_core_tests PRIVATE astra_core Threads::Threads)
    target_compile_options(astra_core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME astra_core_tests COMMAND astra_core_tests)
    add_executable(astra_synth_tests tests/SynthTests.cpp tests/AllocationTracker.cpp)
    target_link_libraries(astra_synth_tests PRIVATE astra_core Threads::Threads)
    target_compile_options(astra_synth_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME astra_synth_tests COMMAND astra_synth_tests)
    add_executable(astra_fx_tests tests/FxTests.cpp tests/AllocationTracker.cpp)
    target_link_libraries(astra_fx_tests PRIVATE astra_core Threads::Threads)
    target_compile_options(astra_fx_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME astra_fx_tests COMMAND astra_fx_tests)
    add_executable(astra_render_presets tests/RenderPresets.cpp)
    target_link_libraries(astra_render_presets PRIVATE astra_core)
    target_compile_options(astra_render_presets PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
