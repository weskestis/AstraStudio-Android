cmake_minimum_required(VERSION 3.22.1)
project(astra_audio LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(astra_core STATIC core/RenderCore.cpp sampler/Sampler.cpp dsp/SynthPatch.cpp dsp/WavetableBank.cpp dsp/ImportedWavetable.cpp dsp/Filter.cpp dsp/SynthVoice.cpp
    fx/FxConfig.cpp fx/FxPrimitives.cpp fx/FxModules.cpp fx/FxRack.cpp fx/Spectrum.cpp project/Project.cpp project/OfflineRender.cpp project/TempoMap.cpp project/WaveWriter.cpp project/StretchRender.cpp project/WarpVoice.cpp project/SampleStream.cpp)
set_source_files_properties(dsp/ImportedWavetable.cpp project/Project.cpp project/OfflineRender.cpp project/StretchRender.cpp project/WarpVoice.cpp project/SampleStream.cpp PROPERTIES COMPILE_OPTIONS "-fexceptions")
set_target_properties(astra_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(astra_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(astra_core SYSTEM PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/third_party)
target_compile_options(astra_core PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions)
target_compile_options(astra_core PRIVATE $<$<CONFIG:Debug>:-O2>)

if(ANDROID)
    target_compile_options(astra_core PRIVATE -fno-rtti)
    find_package(oboe REQUIRED CONFIG)
    add_library(astra_audio SHARED AudioEngine.cpp native-lib.cpp)
    target_link_libraries(astra_audio PRIVATE astra_core oboe::oboe android log)
    target_compile_options(astra_audio PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(astra_audio PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
